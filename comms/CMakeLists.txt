cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(comms C CXX ASM)
pico_sdk_init()

set(CLOUD http CACHE STRING "cloud backend: http or serial")
set(CLOUD_SRC cloud_${CLOUD}.c)

add_executable(comms main.c app.c cmd.c sched.c ui.c ${CLOUD_SRC} ../common/protocol.c ../common/link.c)
target_include_directories(comms PRIVATE ${CMAKE_CURRENT_LIST_DIR} ../common)
target_compile_definitions(comms PRIVATE BOARD_COMMS)
target_link_libraries(comms pico_stdlib hardware_uart hardware_gpio pico_lwip_sntp pico_cyw43_arch_lwip_threadsafe_background)
pico_enable_stdio_usb(comms 1)
pico_enable_stdio_uart(comms 0)
pico_add_extra_outputs(comms)
