cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(control C CXX ASM)
pico_sdk_init()

add_executable(control main.c hw.c ctl.c fsm.c lift.c ../common/protocol.c ../common/link.c)
target_include_directories(control PRIVATE ${CMAKE_CURRENT_LIST_DIR} ../common)
target_compile_definitions(control PRIVATE BOARD_CONTROL)
option(LOG_IMU "print accelerometer values over USB" OFF)
if(LOG_IMU)
    target_compile_definitions(control PRIVATE LOG_IMU)
endif()
target_link_libraries(control pico_stdlib hardware_uart hardware_pwm hardware_i2c hardware_gpio hardware_clocks m)
pico_enable_stdio_usb(control 1)
pico_enable_stdio_uart(control 0)
pico_add_extra_outputs(control)
